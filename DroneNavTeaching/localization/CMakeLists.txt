cmake_minimum_required(VERSION 3.16)
project(drone_orb_ipc)
set(CMAKE_CXX_STANDARD 14)
find_package(OpenCV REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Pangolin REQUIRED)
set(ORB_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../vendor/ORB_SLAM3)
include_directories(${ORB_ROOT} ${ORB_ROOT}/include ${ORB_ROOT}/include/CameraModels ${ORB_ROOT}/Thirdparty/Sophus ${EIGEN3_INCLUDE_DIR} ${Pangolin_INCLUDE_DIRS})
link_directories(${ORB_ROOT}/lib)
add_executable(orb_ipc orb_ipc.cpp)
target_link_libraries(orb_ipc ORB_SLAM3 ${OpenCV_LIBS} ${Pangolin_LIBRARIES})
set_target_properties(orb_ipc PROPERTIES BUILD_RPATH "${ORB_ROOT}/lib;${CMAKE_CURRENT_SOURCE_DIR}/../vendor/pangolin-install/lib")
